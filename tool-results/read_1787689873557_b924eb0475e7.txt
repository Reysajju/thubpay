   201→export async function getInvoiceViewStats(workspaceId: string) {
   202→  if (workspaceId === 'ws-demo-workspace') {
   203→    return { sentCount: 0, viewedCount: 0, notViewedCount: 0, openRate: 0, recentOpens: [] };
   204→  }
   205→  try {
   206→    const [sentCount, viewedCount, openedRows] = await Promise.all([
   207→      db.invoice.count({ where: { workspaceId, status: { in: ['sent', 'viewed', 'paid', 'overdue'] } } }),
   208→      db.invoice.count({ where: { workspaceId, firstViewedAt: { not: null } } }),
   209→      db.invoice.findMany({
   210→        where: { workspaceId, firstViewedAt: { not: null } },
   211→        orderBy: { lastViewedAt: 'desc' },
   212→        take: 10,
   213→        include: { client: true },
   214→      }),
   215→    ]);
   216→
   217→    const recentOpens = openedRows.map((r) => ({
   218→      id: r.id,
   219→      invoice_number: r.invoiceNumber,
   220→      client_name: r.client?.name ?? null,
   221→      client_email: r.client?.email ?? null,
   222→      first_viewed_at: r.firstViewedAt ? r.firstViewedAt.toISOString() : null,
   223→      last_viewed_at: r.lastViewedAt ? r.lastViewedAt.toISOString() : null,
   224→      view_count: r.viewCount,
   225→      total_cents: r.totalCents,
   226→      status: r.status,
   227→      last_viewer_location: r.lastViewerLocation,
   228→    }));
   229→
   230→    const openRate = sentCount > 0 ? Math.round((viewedCount / sentCount) * 100) : 0;
   231→
   232→    return {
   233→      sentCount,
   234→      viewedCount,
   235→      notViewedCount: Math.max(0, sentCount - viewedCount),
   236→      openRate,
   237→      recentOpens,
   238→    };
   239→  } catch {
   240→    return { sentCount: 0, viewedCount: 0, notViewedCount: 0, openRate: 0, recentOpens: [] };
   241→  }
   242→}

   400→export async function getTransactions(
   401→  workspaceId: string,
   402→  options?: {
   403→    status?: string;
   404→    gatewaySlug?: string;
   405→    limit?: number;
   406→    startDate?: Date;
   407→    endDate?: Date;
   408→  }
   409→): Promise<DemoTransaction[]> {
   410→  const where: any = { workspaceId };
   411→  if (options?.status && options.status !== 'all') {
   412→    where.status = options.status;
   413→  }
   414→  if (options?.gatewaySlug && options.gatewaySlug !== 'all') {
   415→    where.gatewaySlug = options.gatewaySlug;
   416→  }
   417→  if (options?.startDate || options?.endDate) {
   418→    where.createdAt = {};
   419→    if (options?.startDate) where.createdAt.gte = options.startDate;
   420→    if (options?.endDate) where.createdAt.lte = options.endDate;
   421→  }
   422→
   423→  const rows = await db.transaction.findMany({
   424→    where,
   425→    orderBy: { createdAt: 'desc' },
   426→    take: options?.limit ?? 200,
   427→    include: { gateway: true, invoice: true },
   428→  });
   429→
   430→  return rows.map((t) => ({
   431→    id: t.id,
   432→    workspace_id: t.workspaceId,
   433→    invoice_id: t.invoiceId,
   434→    gateway_id: t.gatewayId,
   435→    gateway_slug: t.gatewaySlug,
   436→    gateway_label: t.gateway?.label || null,
   437→    external_id: t.externalId,
   438→    amount_cents: t.amountCents,
   439→    currency: t.currency,
   440→    status: t.status as DemoTransaction['status'],
   441→    failure_reason: t.failureReason,
   442→    customer_email: t.customerEmail,
   443→    customer_name: t.customerName,
   444→    invoice_number: t.invoice?.invoiceNumber || null,
   445→    created_at: t.createdAt.toISOString(),
   446→    updated_at: t.updatedAt.toISOString(),
   447→  }));

   450→export async function getTransactionStats(workspaceId: string) {
   451→  const transactions = await db.transaction.findMany({
   452→    where: { workspaceId },
   453→    select: {
   454→      amountCents: true,
   455→      status: true,
   456→      gatewaySlug: true,
   457→      createdAt: true,
   458→    },
   459→  });
   460→
   461→  const totalVolume = transactions
   462→    .filter((t) => t.status === 'succeeded')
   463→    .reduce((s, t) => s + t.amountCents, 0);
   464→  const succeededCount = transactions.filter((t) => t.status === 'succeeded').length;
   465→  const failedCount = transactions.filter((t) => t.status === 'failed').length;
   466→  const pendingCount = transactions.filter((t) => t.status === 'pending').length;
   467→  const refundedCount = transactions.filter((t) => t.status === 'refunded').length;
   468→  const disputedCount = transactions.filter((t) => t.status === 'disputed').length;
   469→  const totalCount = transactions.length;
   470→  const successRate =
   471→    totalCount > 0 ? Math.round((succeededCount / totalCount) * 100) : 0;
   472→
   473→  // Average transaction value (succeeded only)
   474→  const avgAmount =
   475→    succeededCount > 0
   476→      ? Math.round(totalVolume / succeededCount)
   477→      : 0;
   478→
   479→  // Group by gateway
   480→  const byGateway: Record<string, { count: number; volume: number }> = {};
   481→  for (const t of transactions) {
   482→    if (!byGateway[t.gatewaySlug]) {
   483→      byGateway[t.gatewaySlug] = { count: 0, volume: 0 };
   484→    }
   485→    byGateway[t.gatewaySlug].count += 1;
   486→    if (t.status === 'succeeded') {
   487→      byGateway[t.gatewaySlug].volume += t.amountCents;
   488→    }
   489→  }
   490→
   491→  return {
   492→    totalVolume,
   493→    succeededCount,
   494→    failedCount,
   495→    pendingCount,
   496→    refundedCount,
   497→    disputedCount,
   498→    totalCount,
   499→    successRate,
   500→    avgAmount,
   501→    byGateway,
   502→  };
   503→}

   505→// ─── Transaction Volume (last 30 days) ────────────────────────
   506→
   507→export async function getTransactionVolume(workspaceId: string, days = 30) {
   508→  const now = new Date();
   509→  const startDate = new Date(now);
   510→  startDate.setDate(startDate.getDate() - (days - 1));
   511→  startDate.setHours(0, 0, 0, 0);
   512→
   513→  const transactions = await db.transaction.findMany({
   514→    where: {
   515→      workspaceId,
   516→      createdAt: { gte: startDate },
   517→    },
   518→    select: {
   519→      amountCents: true,
   520→      status: true,
   521→      createdAt: true,
   522→    },
   523→  });
   524→
   525→  const data: { date: string; volume: number; count: number; failed: number }[] = [];
   526→  for (let i = days - 1; i >= 0; i--) {
   527→    const d = new Date(now);
   528→    d.setDate(d.getDate() - i);
   529→    d.setHours(0, 0, 0, 0);
   530→    const dayEnd = new Date(d);
   531→    dayEnd.setHours(23, 59, 59, 999);
   532→
   533→    const dayTxns = transactions.filter(
   534→      (t) => t.createdAt >= d && t.createdAt <= dayEnd
   535→    );
   536→
   537→    const label = d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   538→    const volume = dayTxns
   539→      .filter((t) => t.status === 'succeeded')
   540→      .reduce((s, t) => s + t.amountCents, 0);
   541→    const count = dayTxns.filter((t) => t.status === 'succeeded').length;
   542→    const failed = dayTxns.filter((t) => t.status === 'failed').length;
   543→
   544→    data.push({ date: label, volume, count, failed });
   545→  }
   546→
   547→  return data;
   548→}

   550→// ─── Analytics: Monthly Revenue (last 12 months) ──────────────
   551→
   552→export async function getMonthlyRevenue(workspaceId: string) {
   553→  const now = new Date();
   554→  const data: { date: string; amount: number }[] = [];
   555→
   556→  // Pull all paid transactions and group by month.
   557→  const txns = await db.transaction.findMany({
   558→    where: { workspaceId, status: 'succeeded' },
   559→    select: { amountCents: true, createdAt: true },
   560→  });
   561→
   562→  for (let i = 11; i >= 0; i--) {
   563→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   564→    const label = d.toLocaleDateString('en-US', {
   565→      month: 'short',
   566→      year: '2-digit',
   567→    });
   568→    const monthStart = new Date(d.getFullYear(), d.getMonth(), 1);
   569→    const monthEnd = new Date(d.getFullYear(), d.getMonth() + 1, 1);
   570→    const amount = txns
   571→      .filter((t) => t.createdAt >= monthStart && t.createdAt < monthEnd)
   572→      .reduce((s, t) => s + t.amountCents, 0);
   573→    data.push({ date: label, amount });
   574→  }
   575→
   576→  // If no real transactions yet, fall back to invoice-derived revenue
   577→  // so the dashboard chart isn't empty on a fresh workspace.
   578→  const totalFromData = data.reduce((s, d) => s + d.amount, 0);
   579→  if (totalFromData === 0) {
   580→    const paidInvoices = await db.invoice.findMany({
   581→      where: { workspaceId, status: 'paid' },
   582→      select: { totalCents: true, createdAt: true },
   583→    });
   584→    for (let i = 0; i < data.length; i++) {
   585→      const d = new Date(now.getFullYear(), now.getMonth() - (11 - i), 1);
   586→      const monthStart = new Date(d.getFullYear(), d.getMonth(), 1);

   702→export interface LedgerEntry {
   703→  id: string;
   704→  direction: 'incoming' | 'outgoing';
   705→  amount_cents: number;
   706→  note: string;

   712→export interface FinanceData {
   713→  totalIncoming: number;
   714→  totalOutgoing: number;
   715→  totalFees: number;
   716→  netCash: number;
   717→  grossRevenue: number;
   718→  outstanding: number;
   719→  overdue: number;
   720→  ledger: LedgerEntry[];
   721→  monthlyRevenue: { month: string; revenue: number; fees: number }[];
   722→  feeBreakdown: { gateway: string; fee: number; count: number }[];
   723→}
   724→
   725→// Processing fee rates per gateway (typical rates)
   726→const FEE_RATES: Record<string, { percent: number; fixed: number }> = {
   727→  stripe: { percent: 0.029, fixed: 30 },
   728→  paypal: { percent: 0.0349, fixed: 49 },
   729→  square: { percent: 0.026, fixed: 10 },
   730→  adyen: { percent: 0.026, fixed: 12 },
   731→  razorpay: { percent: 0.020, fixed: 0 },
   732→  authorize_net: { percent: 0.029, fixed: 30 },
   733→  braintree: { percent: 0.029, fixed: 30 },
   734→  mollie: { percent: 0.029, fixed: 25 },
   735→  manual: { percent: 0, fixed: 0 },
   736→  custom: { percent: 0.029, fixed: 30 },
   737→};
   738→
   739→export async function getFinanceData(workspaceId: string): Promise<FinanceData> {
   740→  const [transactions, invoices] = await Promise.all([
   741→    db.transaction.findMany({
   742→      where: { workspaceId },
   743→      orderBy: { createdAt: 'desc' },
   744→      take: 200,
   745→      include: { gateway: true, invoice: true },
   746→    }),
   747→    db.invoice.findMany({
   748→      where: { workspaceId },
   749→      select: {
   750→        totalCents: true,
   751→        status: true,
   752→        createdAt: true,
   753→        updatedAt: true,
   754→        paidViaGateway: true,
   755→        customPaymentGateway: true,
   756→      },
   757→    }),
   758→  ]);
   759→
   760→  // Build ledger from transactions
   761→  const ledger: LedgerEntry[] = [];
   762→
   763→  for (const tx of transactions) {
   764→    if (tx.status === 'succeeded') {
   765→      // Incoming: payment received
   766→      ledger.push({
   767→        id: `led-in-${tx.id}`,
   768→        direction: 'incoming',
   769→        amount_cents: tx.amountCents,
   770→        note: tx.invoice?.invoiceNumber
   771→          ? `Payment for ${tx.invoice.invoiceNumber}`
   772→          : `Payment from ${tx.customerName || tx.customerEmail || 'customer'}`,
   773→        occurred_at: tx.createdAt.toISOString(),
   774→        type: 'payment',
   775→        gateway: tx.gatewaySlug,
   776→      });
   777→
   778→      // Outgoing: processing fee
   779→      const rate = FEE_RATES[tx.gatewaySlug] || FEE_RATES.custom;
   780→      const fee = Math.round(tx.amountCents * rate.percent + rate.fixed);
   781→      if (fee > 0) {
   782→        ledger.push({
   783→          id: `led-fee-${tx.id}`,
   784→          direction: 'outgoing',
   785→          amount_cents: fee,
   786→          note: `${tx.gatewaySlug} processing fee`,
   787→          occurred_at: tx.createdAt.toISOString(),
   788→          type: 'fee',
   789→          gateway: tx.gatewaySlug,
   790→        });
   791→      }
   792→    } else if (tx.status === 'refunded') {
   793→      ledger.push({
   794→        id: `led-ref-${tx.id}`,
   795→        direction: 'outgoing',
   796→        amount_cents: tx.amountCents,
   797→        note: `Refund for ${tx.invoice?.invoiceNumber || tx.externalId || 'transaction'}`,
   798→        occurred_at: tx.updatedAt.toISOString(),
   799→        type: 'refund',
   800→        gateway: tx.gatewaySlug,
   801→      });
   802→    }
   803→  }
   804→
   805→  // Add monthly payout entries (synthesized for demo)
   806→  const now = new Date();
   807→  for (let i = 0; i < 3; i++) {
   808→    const payoutDate = new Date(now.getFullYear(), now.getMonth() - i, 1);
   809→    const monthRevenue = transactions
   810→      .filter((t) => {
   811→        if (t.status !== 'succeeded') return false;
   812→        const d = t.createdAt;
   813→        return d.getMonth() === payoutDate.getMonth() && d.getFullYear() === payoutDate.getFullYear();
   814→      })
   815→      .reduce((s, t) => s + t.amountCents, 0);
   816→    if (monthRevenue > 0) {
   817→      const payoutAmount = Math.round(monthRevenue * 0.7); // 70% payout
   818→      ledger.push({
   819→        id: `led-payout-${i}`,
   820→        direction: 'outgoing',
   821→        amount_cents: payoutAmount,
   822→        note: `Monthly payout to bank account`,
   823→        occurred_at: payoutDate.toISOString(),
   824→        type: 'payout',
   825→      });
   826→    }
   827→  }
   828→
   829→  // Sort ledger by date descending
   830→  ledger.sort((a, b) => new Date(b.occurred_at).getTime() - new Date(a.occurred_at).getTime());
   831→
   832→  const totalIncoming = ledger
   833→    .filter((e) => e.direction === 'incoming')
   834→    .reduce((s, e) => s + e.amount_cents, 0);
   835→  const totalOutgoing = ledger
   836→    .filter((e) => e.direction === 'outgoing')
   837→    .reduce((s, e) => s + e.amount_cents, 0);
   838→  const totalFees = ledger
   839→    .filter((e) => e.type === 'fee')
   840→    .reduce((s, e) => s + e.amount_cents, 0);
   841→  const netCash = totalIncoming - totalOutgoing;
   842→
   843→  const grossRevenue = invoices
   844→    .filter((i) => i.status === 'paid')
   845→    .reduce((s, i) => s + i.totalCents, 0);
   846→  const outstanding = invoices
   847→    .filter((i) => ['sent', 'viewed'].includes(i.status))
   848→    .reduce((s, i) => s + i.totalCents, 0);
   849→  const overdue = invoices
   850→    .filter((i) => i.status === 'overdue')
   851→    .reduce((s, i) => s + i.totalCents, 0);
   852→
   853→  // Monthly revenue + fees trend (last 6 months)
   854→  const monthlyRevenue: { month: string; revenue: number; fees: number }[] = [];
   855→  for (let i = 5; i >= 0; i--) {
   856→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   857→    const monthLabel = d.toLocaleDateString('en-US', { month: 'short' });
   858→    const monthStart = new Date(d.getFullYear(), d.getMonth(), 1);
   859→    const monthEnd = new Date(d.getFullYear(), d.getMonth() + 1, 1);
   860→
   861→    const monthTxns = transactions.filter(
   862→      (t) => t.status === 'succeeded' && t.createdAt >= monthStart && t.createdAt < monthEnd
   863→    );
   864→    const revenue = monthTxns.reduce((s, t) => s + t.amountCents, 0);
   865→    const fees = monthTxns.reduce((s, t) => {
   866→      const rate = FEE_RATES[t.gatewaySlug] || FEE_RATES.custom;
   867→      return s + Math.round(t.amountCents * rate.percent + rate.fixed);
   868→    }, 0);
   869→
   870→    monthlyRevenue.push({ month: monthLabel, revenue, fees });
   871→  }
   872→
   873→  // Fee breakdown by gateway
   874→  const feeMap: Record<string, { fee: number; count: number }> = {};
   875→  for (const tx of transactions) {
   876→    if (tx.status !== 'succeeded') continue;
   877→    const rate = FEE_RATES[tx.gatewaySlug] || FEE_RATES.custom;
   878→    const fee = Math.round(tx.amountCents * rate.percent + rate.fixed);
   879→    if (!feeMap[tx.gatewaySlug]) feeMap[tx.gatewaySlug] = { fee: 0, count: 0 };
   880→    feeMap[tx.gatewaySlug].fee += fee;
   881→    feeMap[tx.gatewaySlug].count += 1;
   882→  }
   883→  const feeBreakdown = Object.entries(feeMap)
   884→    .map(([gateway, data]) => ({ gateway, fee: data.fee, count: data.count }))
   885→    .sort((a, b) => b.fee - a.fee);
   886→
   887→  return {
   888→    totalIncoming,
   889→    totalOutgoing,
   890→    totalFees,
   891→    netCash,
   892→    grossRevenue,
   893→    outstanding,
   894→    overdue,
   895→    ledger,
   896→    monthlyRevenue,
   897→    feeBreakdown,
   898→  };
   899→}

  1067→

  1092→    } else {
  1093→      // Sync with actual state — auto-detect completed steps

  1096→        db.gatewayCredential.count({ where: { workspaceId, isActive: true, secretKeyEnc: { not: null } } }),

  1124→    const completionPct = Math.round((completedSteps / 4) * 100);

  1372→  if (endpoints.length === 0) return [];

  1386→      durationMs: true,
  1387→      attemptedAt: true,

  1391→  // Group by endpoint

  1413→    const min = latencies.length > 0 ? latencies[0] : null;
  1414→    const max = latencies.length > 0 ? latencies[latencies.length - 1] : null;
  1415→    const lastAt = eps.length > 0 ? eps[0].attemptedAt.toISOString() : null;
  1416→    return {
  1417→      endpoint_id: ep.id,

  1434→export interface InvoiceHeatmapData {
  1435→  // 7 days × 24 hours = 168 cells, each with the count of views
  1436→  // days: 0=Sun, 1=Mon, ..., 6=Sat
  1437→  // hours: 0-23
  1438→  cells: { day: number; hour: number; count: number }[];
  1439→  total_views: number;
  1440→  peak_day: number | null;
  1441→  peak_hour: number | null;
  1442→  peak_cell: { day: number; hour: number; count: number } | null;
  1443→}

  1445→export async function getInvoiceOpenHeatmap(workspaceId: string): Promise<InvoiceHeatmapData> {
  1446→  // Get all invoice views for the workspace (via the Invoice join)
  1447→  const views = await db.invoiceView.findMany({
  1448→    where: { invoice: { workspaceId } },
  1449→    select: { viewedAt: true },
  1450→    take: 10000,
  1451→  });
  1452→
  1453→  // Build a 7×24 grid (initialized to 0)
  1454→  const grid: number[][] = Array.from({ length: 7 }, () => Array.from({ length: 24 }, () => 0));
  1455→  let totalViews = 0;
  1456→  let peakDay: number | null = null;
  1457→  let peakHour: number | null = null;
  1458→  let peakCell: { day: number; hour: number; count: number } | null = null;
  1459→  let peakCount = 0;
  1460→
  1461→  // Day/hour totals for finding peaks
  1462→  const dayTotals = new Array(7).fill(0);
  1463→  const hourTotals = new Array(24).fill(0);
  1464→
  1465→  for (const v of views) {
  1466→    const d = new Date(v.viewedAt);
  1467→    const day = d.getDay();
  1468→    const hour = d.getHours();
  1469→    grid[day][hour]++;
  1470→    dayTotals[day]++;
  1471→    hourTotals[hour]++;
  1472→    totalViews++;
  1473→    if (grid[day][hour] > peakCount) {
  1474→      peakCount = grid[day][hour];
  1475→      peakCell = { day, hour, count: peakCount };
  1476→    }
  1477→  }
  1478→
  1479→  if (totalViews > 0) {
  1480→    peakDay = dayTotals.indexOf(Math.max(...dayTotals));
  1481→    peakHour = hourTotals.indexOf(Math.max(...hourTotals));
  1482→  }
  1483→
  1484→  // Flatten grid to cells array
  1485→  const cells: { day: number; hour: number; count: number }[] = [];
  1486→  for (let day = 0; day < 7; day++) {
  1487→    for (let hour = 0; hour < 24; hour++) {
  1488→      cells.push({ day, hour, count: grid[day][hour] });
  1489→    }
  1490→  }
  1491→
  1492→  return {
  1493→    cells,
  1494→    total_views: totalViews,
  1495→    peak_day: peakDay,
  1496→    peak_hour: peakHour,
  1497→    peak_cell: peakCell,
  1498→  };
  1499→}
