   107→function timeAgo(iso: string): string {
   108→  const diff = Date.now() - new Date(iso).getTime();
   109→  const mins = Math.floor(diff / 60000);
   110→  if (mins < 1) return 'just now';
   111→  if (mins < 60) return `${mins}m ago`;
   112→  const hours = Math.floor(mins / 60);
   113→  if (hours < 24) return `${hours}h ago`;
   114→  const days = Math.floor(hours / 24);
   115→  if (days < 7) return `${days}d ago`;
   116→  return new Date(iso).toLocaleDateString();
   117→}
