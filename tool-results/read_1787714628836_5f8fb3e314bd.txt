   150→function formatNumber(num: number) {
   151→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`;

   153→  return num.toString();

   156→function getAvatarGradient(seed: string): string {
   157→  const gradients = [
   158→    'from-amber-500/20 to-orange-600/20 text-amber-400',
   159→    'from-emerald-500/20 to-teal-600/20 text-emerald-400',
   160→    'from-cyan-500/20 to-teal-600/20 text-cyan-400',
   161→    'from-purple-500/20 to-pink-600/20 text-purple-400',
   162→    'from-rose-500/20 to-red-600/20 text-rose-400',
   163→    'from-cyan-500/20 to-sky-600/20 text-cyan-400',
   164→  ];
   165→  let hash = 0;
   166→  for (let i = 0; i < seed.length; i++) {
   167→    hash = seed.charCodeAt(i) + ((hash << 5) - hash);
   168→  }
   169→  return gradients[Math.abs(hash) % gradients.length];
   170→}
