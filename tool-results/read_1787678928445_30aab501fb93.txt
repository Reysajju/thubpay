    11→function toUsd(cents: number) {
    12→  return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format((cents || 0) / 100);
    13→}
    14→
    15→function formatDate(date: Date | string | null): string {
    16→  if (!date) return '—';
    17→  const d = typeof date === 'string' ? new Date(date) : date;
    18→  return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
    19→}
    20→
    21→function formatDateTime(date: Date | string | null): string {
    22→  if (!date) return '—';
    23→  const d = typeof date === 'string' ? new Date(date) : date;
    24→  return d.toLocaleString('en-US', {
    25→    year: 'numeric',
    26→    month: 'short',
    27→    day: 'numeric',
    28→    hour: '2-digit',
    29→    minute: '2-digit',
    30→  });
    31→}
