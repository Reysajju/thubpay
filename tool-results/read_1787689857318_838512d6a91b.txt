   102→function formatDate(dateString: string) {

   109→function formatNumber(num: number) {
   110→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`;
   111→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`;
   112→  return num.toString();
   113→}

   115→function getAvatarGradient(seed: string): string {
   116→  const gradients = [
   117→    'from-amber-500/20 to-orange-600/20 text-amber-400',
   118→    'from-emerald-500/20 to-teal-600/20 text-emerald-400',
   119→    'from-blue-500/20 to-indigo-600/20 text-blue-400',
   120→    'from-purple-500/20 to-pink-600/20 text-purple-400',
   121→    'from-rose-500/20 to-red-600/20 text-rose-400',
   122→    'from-cyan-500/20 to-sky-600/20 text-cyan-400',
   123→  ];
   124→  let hash = 0;
   125→  for (let i = 0; i < seed.length; i++) {
   126→    hash = seed.charCodeAt(i) + ((hash << 5) - hash);
   127→  }
   128→  return gradients[Math.abs(hash) % gradients.length];
   129→}
