     9→     9→// ─── Workspace context ────────────────────────────────────────
    10→    10→
    11→    11→async function getWorkspaceContext() {
    12→    12→  const session = await auth();
    13→    13→  if (!session?.user) return null;
    14→    14→
    15→    15→  const userId = (session.user as any).id as string | undefined;
    16→    16→  if (!userId) return null;
    17→    17→
    18→    18→  const membership = await db.workspaceMember.findFirst({
    19→    19→    where: { userId },
    20→    20→    include: { workspace: true },
    21→    21→    orderBy: { createdAt: 'asc' },
    22→    22→  });
    23→    23→  if (!membership) return null;
    24→    24→
    25→    25→  return {
    26→    26→    user: {
    27→    27→      id: userId,
    28→    28→      email: session.user.email || '',
    29→    29→      name: session.user.name || '',
    30→    30→      role: membership.role || 'owner',
    31→    31→    },
    32→    32→    workspaceId: membership.workspaceId,
    33→    33→    workspace: membership.workspace,
    34→    34→  };
    35→    35→}

    94→    94→// ─── CREATE INVOICE ────────────────────────────────────────────
    95→    95→
    96→    96→export async function createInvoice(formData: FormData) {
    97→    97→  const ctx = await getWorkspaceContext();
    98→    98→  if (!ctx) return;
    99→    99→
   100→   100→  const clientId = String(formData.get('client_id') ?? '').trim();
   101→   101→  const amount = Number(formData.get('total_usd') ?? 0);
   102→   102→  const taxRate = Number(formData.get('tax_rate_pct') ?? 0);
   103→   103→  const dueDateRaw = String(formData.get('due_date') ?? '').trim();
   104→   104→  const notes = String(formData.get('notes') ?? '').trim() || null;
   105→   105→  const description = String(formData.get('description') ?? '').trim();
   106→   106→  const gatewaySlug = String(formData.get('gateway_slug') ?? '').trim() || null;
   107→   107→
   108→   108→  if (!clientId || !Number.isFinite(amount) || amount <= 0) {
   109→   109→    return;
   110→   110→  }
   111→   111→
   112→   112→  const subtotalCents = Math.round(amount * 100);
   113→   113→  const taxCents = Math.round(subtotalCents * (taxRate / 100));
   114→   114→  const totalCents = subtotalCents + taxCents;
   115→   115→
   116→   116→  // Generate invoice number: INV-YYYY-NNN
   117→   117→  const year = new Date().getFullYear();
   118→   118→  const count = await db.invoice.count({
   119→   119→    where: {
   120→   120→      workspaceId: ctx.workspaceId,
   121→   121→      invoiceNumber: { startsWith: `INV-${year}-` },
   122→   122→    },
   123→   123→  });
   124→   124→  const invoiceNumber = `INV-${year}-${String(count + 1).padStart(3, '0')}`;
   125→   125→
   126→   126→  const invoice = await db.invoice.create({
   127→   127→    data: {
   128→   128→      workspaceId: ctx.workspaceId,
   129→   129→      clientId,
   130→   130→      invoiceNumber,
   131→   131→      status: 'draft',
   132→   132→      totalCents,
   133→   133→      currency: ctx.workspace.baseCurrency || 'USD',
   134→   134→      dueDate: dueDateRaw ? new Date(dueDateRaw) : null,
   135→   135→      notes: notes || description || null,
   136→   136→      customPaymentGateway: gatewaySlug || null,
   137→   137→      // Generate the link-tracking token at creation time so the tracking
   138→   138→      // pixel can fire the moment the invoice is dispatched.
   139→   139→      trackingToken: await import('crypto').then((c) => c.randomBytes(16).toString('hex')),
   140→   140→    },
   141→   141→  });
   142→   142→
   143→   143→  // Auto-complete onboarding step
   144→   144→  await db.onboardingProgress.upsert({
   145→   145→    where: { workspaceId: ctx.workspaceId },
   146→   146→    update: { stepInvoice: true },
   147→   147→    create: { workspaceId: ctx.workspaceId, stepInvoice: true },
   148→   148→  });
   149→   149→
   150→   150→  revalidatePath('/dashboard');
   151→   151→  revalidatePath('/dashboard/transactions');
   152→   152→  redirect(`/invoice/${invoice.id}`);
   153→   153→}

   155→   155→// ─── DISPATCH INVOICE (mark as sent) ──────────────────────────
   156→   156→
   157→   157→export async function dispatchInvoice(formData: FormData) {
   158→   158→  const ctx = await getWorkspaceContext();
   159→   159→  if (!ctx) return;
   160→   160→
   161→   161→  const invoiceId = String(formData.get('invoice_id') ?? '').trim();
   162→   162→  if (!invoiceId) return;
   163→   163→
   164→   164→  // Verify the invoice belongs to this workspace before mutating.
   165→   165→  const invoice = await db.invoice.findFirst({
   166→   166→    where: { id: invoiceId, workspaceId: ctx.workspaceId },
   167→   167→    include: { client: true, workspace: true },
   168→   168→  });
   169→   169→  if (!invoice) return;
   170→   170→
   171→   171→  // Generate a tracking token if the invoice doesn't have one yet.
   172→   172→  const trackingToken =
   173→   173→    invoice.trackingToken ??
   174→   174→    (await import('crypto').then((c) => c.randomBytes(16).toString('hex')));
   175→   175→
   176→   176→  if (invoice.status === 'draft') {
   177→   177→    await db.invoice.update({
   178→   178→      where: { id: invoiceId },
   179→   179→      data: {
   180→   180→        status: 'sent',
   181→   181→        sentAt: invoice.sentAt ?? new Date(),
   182→   182→        trackingToken,
   183→   183→      },
   184→   184→    });
   185→   185→  } else if (!invoice.trackingToken || !invoice.sentAt) {
   186→   186→    await db.invoice.update({
   187→   187→      where: { id: invoiceId },
   188→   188→      data: {
   189→   189→        sentAt: invoice.sentAt ?? new Date(),
   190→   190→        trackingToken,
   191→   191→      },
   192→   192→    });
   193→   193→  }
   194→   194→
   195→   195→  // Send real email if client email is available
   196→   196→  if (invoice.client?.email) {
   197→   197→    const { sendInvoiceEmail } = await import('@/lib/email');
   198→   198→    const paymentUrl = `${process.env.NEXTAUTH_URL || 'http://localhost:3000'}/pay/${invoice.id}`;
   199→   199→    await sendInvoiceEmail({
   200→   200→      to: invoice.client.email,
   201→   201→      invoiceNumber: invoice.invoiceNumber || invoice.id.slice(0, 8),
   202→   202→      amountFormatted: `$${((invoice.totalCents || 0) / 100).toFixed(2)}`,
   203→   203→      paymentUrl,
   204→   204→      clientName: invoice.client.name,
   205→   205→      businessName: invoice.workspace?.name || 'ThubPay',
   206→   206→    }).catch((err) => console.error('[dispatchInvoice email error]:', err));
   207→   207→  }
   208→   208→
   209→   209→  // Record a notification with payment link
   210→   210→  const paymentLink = `/pay/${invoiceId}`;
   211→   211→  await db.notification.create({
   212→   212→    data: {
   213→   213→      workspaceId: ctx.workspaceId,
   214→   214→      title: 'Invoice sent to client',
   215→   215→      body: `${invoice.invoiceNumber || invoiceId} has been dispatched. Payment link: ${paymentLink}`,
   216→   216→      type: 'success',
   217→   217→    },
   218→   218→  });
   219→   219→
   220→   220→  revalidatePath(`/invoice/${invoiceId}`);
   221→   221→  revalidatePath('/dashboard');
   222→   222→  revalidatePath('/dashboard/transactions');
   223→   223→  revalidatePath('/dashboard/link-tracking');
   224→   224→}

   391→   391→export async function markInvoicePaidManually(invoiceId: string) {
   392→   392→  const ctx = await getWorkspaceContext();
   393→   393→  if (!ctx) return { success: false, error: 'Unauthorized' };
   394→   394→
   395→   395→  const invoice = await db.invoice.findFirst({
   396→   396→    where: { id: invoiceId, workspaceId: ctx.workspaceId },
   397→   397→    include: { client: true },
   398→   398→  });
   399→   399→  if (!invoice) return { success: false, error: 'Invoice not found' };
   400→   400→
   401→   401→  if (invoice.status === 'paid') {
   402→   402→    return { success: true, alreadyPaid: true };
   403→   403→  }
   404→   404→
   405→   405→  await db.$transaction(async (prisma) => {
   406→   406→    await prisma.invoice.update({
   407→   407→      where: { id: invoiceId },
   408→   408→      data: {
   409→   409→        status: 'paid',
   410→   410→        paidAt: new Date(),
   411→   411→        paidViaGateway: invoice.paidViaGateway || 'manual',
   412→   412→        customPaymentGateway: invoice.customPaymentGateway || 'Manual',
   413→   413→      },
   414→   414→    });
   415→   415→
   416→   416→    // Update client aggregates
   417→   417→    if (invoice.clientId) {
   418→   418→      await prisma.client.update({
   419→   419→        where: { id: invoice.clientId },
   420→   420→        data: {
   421→   421→          totalSpendCents: { increment: invoice.totalCents },
   422→   422→          transactionCount: { increment: 1 },
   423→   423→        },
   424→   424→      });
   425→   425→    }
   426→   426→
   427→   427→    // Record a transaction for analytics
   428→   428→    await prisma.transaction.create({
   429→   429→      data: {
   430→   430→        workspaceId: ctx.workspaceId,
   431→   431→        invoiceId,
   432→   432→        gatewaySlug: 'manual',
   433→   433→        externalId: `manual_${Date.now()}`,
   434→   434→        amountCents: invoice.totalCents,
   435→   435→        currency: invoice.currency,
   436→   436→        status: 'succeeded',
   437→   437→        customerEmail: invoice.client?.email || null,
   438→   438→        customerName: invoice.client?.name || null,
   439→   439→      },
   440→   440→    });
   441→   441→
   442→   442→    // Notification
   443→   443→    await prisma.notification.create({
   444→   444→      data: {
   445→   445→        workspaceId: ctx.workspaceId,
   446→   446→        title: 'Payment recorded',
   447→   447→        body: `${invoice.invoiceNumber || invoiceId} marked as paid manually ($${(invoice.totalCents / 100).toFixed(2)}).`,
   448→   448→        type: 'payment',
   449→   449→      },
   450→   450→    });
   451→   451→
   452→   452→    // Audit log entry for the manual mark-paid
   453→   453→    await prisma.auditLog.create({
   454→   454→      data: {
   455→   455→        workspaceId: ctx.workspaceId,
   456→   456→        userId: ctx.user.id,
   457→   457→        action: 'invoice.mark_paid',
   458→   458→        entity: 'invoice',
   459→   459→        entityId: invoiceId,
   460→   460→        metadata: JSON.stringify({
   461→   461→          invoiceNumber: invoice.invoiceNumber,
   462→   462→          amountCents: invoice.totalCents,
   463→   463→          currency: invoice.currency,
   464→   464→          method: 'manual',
   465→   465→          clientId: invoice.clientId,
   466→   466→        }),
   467→   467→      },
   468→   468→    });
   469→   469→  });
   470→   470→
   471→   471→  revalidatePath('/dashboard');
   472→   472→  revalidatePath(`/invoice/${invoiceId}`);
   473→   473→  revalidatePath('/dashboard/transactions');
   474→   474→  revalidatePath('/dashboard/customers');
   475→   475→  revalidatePath('/dashboard/audit-log');
   476→   476→  return { success: true };
   477→   477→}

   598→   598→  }
   599→   599→

   612→   612→    include: { client: true },
   613→   613→  });
   614→   614→

   616→   616→    // Already void — skip.

   622→   622→    if (invoice.status === 'paid') {
   623→   623→      skipped++;

   625→   625→      continue;
   626→   626→    }

   629→   629→      await db.invoice.update({
   630→   630→        where: { id: invoice.id },
   631→   631→        data: { status: 'void' },
   632→   632→      });

   634→   634→      // Email the customer (best-effort, non-blocking).
   635→   635→      const customerEmail = invoice.client?.email || '';
   636→   636→      if (customerEmail) {

   643→   643→
   644→   644→        import('@/lib/email')

   646→   646→            m.sendInvoiceVoidedEmail({
   647→   647→              to: customerEmail,
   648→   648→              clientName: invoice.client?.name || undefined,
   649→   649→              invoiceNumber,

   655→   655→          .catch(() => {

   659→   659→      voided++;
   660→   660→    } catch (err) {
   661→   661→      errors.push(
   662→   662→        `${invoice.invoiceNumber || invoice.id.slice(0, 8)}: ${err instanceof Error ? err.message : 'unknown error'}`
