    32→function toUsd(cents: number) {
    33→  return new Intl.NumberFormat('en-US', {
    34→    style: 'currency',
    35→    currency: 'USD',
    36→    minimumFractionDigits: 2,
    37→  }).format((cents || 0) / 100);
    38→}
    39→
    40→function formatRelative(date: string | Date | null): string | null {
    41→  if (!date) return null;
    42→  const d = typeof date === 'string' ? new Date(date) : date;
    43→  if (Number.isNaN(d.getTime())) return null;
    44→  const diffMs = Date.now() - d.getTime();
    45→  if (diffMs < 0) return 'Just now';
    46→  const sec = Math.floor(diffMs / 1000);
    47→  if (sec < 60) return 'Just now';
    48→  const min = Math.floor(sec / 60);
    49→  if (min < 60) return `${min}m ago`;
    50→  const hr = Math.floor(min / 60);
    51→  if (hr < 24) return `${hr}h ago`;
    52→  const day = Math.floor(hr / 24);
    53→  if (day < 7) return `${day}d ago`;
    54→  const wk = Math.floor(day / 7);
    55→  if (wk < 4) return `${wk}w ago`;
    56→  // Older than ~4 weeks — fall back to a formatted date
    57→  return d.toLocaleDateString(undefined, {
    58→    month: 'short',
    59→    day: 'numeric',
    60→    year: d.getFullYear() === new Date().getFullYear() ? undefined : 'numeric',
    61→  });
    62→}
