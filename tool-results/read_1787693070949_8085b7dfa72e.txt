   149→function formatNumber(num: number) {
   150→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`;

   152→  return num.toString();

   155→function getAvatarGradient(seed: string): string {
   156→  const gradients = [
   157→    'from-amber-500/20 to-orange-600/20 text-amber-400',
   158→    'from-emerald-500/20 to-teal-600/20 text-emerald-400',
   159→    'from-blue-500/20 to-indigo-600/20 text-blue-400',
   160→    'from-purple-500/20 to-pink-600/20 text-purple-400',
   161→    'from-rose-500/20 to-red-600/20 text-rose-400',
   162→    'from-cyan-500/20 to-sky-600/20 text-cyan-400',
   163→  ];
   164→  let hash = 0;
   165→  for (let i = 0; i < seed.length; i++) {
   166→    hash = seed.charCodeAt(i) + ((hash << 5) - hash);
   167→  }
   168→  return gradients[Math.abs(hash) % gradients.length];
   169→}
