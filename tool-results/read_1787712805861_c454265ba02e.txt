   994→function NotificationsTab({ workspaceId }: { workspaceId: string }) {
   995→  const [mutedTypes, setMutedTypes] = useState<string[]>([]);
   996→  const [loading, setLoading] = useState(true);
   997→  const [saving, setSaving] = useState(false);
   998→  const [saved, setSaved] = useState(false);
   999→  const [error, setError] = useState<string | null>(null);
  1000→
  1001→  // Load preferences on mount
  1002→  React.useEffect(() => {
  1003→    let cancelled = false;
  1004→    (async () => {
  1005→      try {
  1006→        const res = (await getNotificationPreferences()) as any;
  1007→        if (!cancelled && res?.success) {
  1008→          setMutedTypes(res.mutedTypes || []);
  1009→        }
  1010→      } catch (err: any) {
  1011→        if (!cancelled) setError(err?.message || 'Failed to load preferences');
  1012→      } finally {
  1013→        if (!cancelled) setLoading(false);
  1014→      }
  1015→    })();
  1016→    return () => { cancelled = true; };
  1017→  }, [workspaceId]);
  1018→
  1019→  function toggleType(type: string) {
  1020→    setMutedTypes((prev) =>
  1021→      prev.includes(type) ? prev.filter((t) => t !== type) : [...prev, type]
  1022→    );
  1023→    setSaved(false);
  1024→  }
  1025→
  1026→  async function handleSave() {
  1027→    setSaving(true);
  1028→    setError(null);
  1029→    setSaved(false);
  1030→    try {
  1031→      const res = (await updateNotificationPreferences(mutedTypes)) as any;
  1032→      if (res?.success) {
  1033→        setSaved(true);
  1034→        setTimeout(() => setSaved(false), 2500);
  1035→      } else {
  1036→        setError(res?.error || 'Failed to save preferences');
  1037→      }
  1038→    } catch (err: any) {
  1039→      setError(err?.message || 'Unexpected error');
  1040→    } finally {
  1041→      setSaving(false);
  1042→    }
  1043→  }
  1044→

  1046→    return (

  1050→      </div>
  1051→    );
  1052→  }
  1053→
  1054→  const activeCount = NOTIFICATION_TYPES.length - mutedTypes.length;
  1055→
  1056→  return (

  1058→      {/* Header card */}

  1064→          <div className="flex-1">
  1065→            <h2 className="text-lg font-bold text-white">Notification Preferences</h2>
  1066→            <p className="text-xs text-zinc-500 mt-0.5">
  1067→              Control which types of notifications you receive. Muted types won&apos;t appear in the bell or the notifications API.
  1068→            </p>
  1069→          </div>
  1070→          <div className="flex items-center gap-2 text-[10px]">
