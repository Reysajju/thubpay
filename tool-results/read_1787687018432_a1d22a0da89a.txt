   753→function NotificationsTab({ workspaceId }: { workspaceId: string }) {
   754→  const [mutedTypes, setMutedTypes] = useState<string[]>([]);

   759→
   760→  // Load preferences on mount

   762→    let cancelled = false;

   764→      try {
   765→        const res = (await getNotificationPreferences()) as any;
   766→        if (!cancelled && res?.success) {
   767→          setMutedTypes(res.mutedTypes || []);
   768→        }
   769→      } catch (err: any) {
   770→        if (!cancelled) setError(err?.message || 'Failed to load preferences');

   796→      }
   797→    } catch (err: any) {
   798→      setError(err?.message || 'Unexpected error');

   800→      setSaving(false);
   801→    }
   802→  }
   803→
   804→  if (loading) {
   805→    return (
   806→      <div className="glass-card rounded-2xl p-8 text-center">

   809→      </div>
   810→    );
   811→  }
   812→
   813→  const activeCount = NOTIFICATION_TYPES.length - mutedTypes.length;
   814→
   815→  return (
   816→    <div className="animate-fadeIn space-y-6">
   817→      {/* Header card */}
   818→      <div className="glass-card rounded-2xl p-5 sm:p-6">

   826→              Control which types of notifications you receive. Muted types won&apos;t appear in the bell or the notifications API.
   827→            </p>
   828→          </div>
   829→          <div className="flex items-center gap-2 text-[10px]">
