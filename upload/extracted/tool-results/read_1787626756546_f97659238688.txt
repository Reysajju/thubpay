    27→function toUsd(cents: number) {
    28→  return new Intl.NumberFormat('en-US', {
    29→    style: 'currency',
    30→    currency: 'USD',
    31→    minimumFractionDigits: 2,
    32→  }).format((cents || 0) / 100);
    33→}
    34→
    35→function formatRelative(date: string | Date | null): string | null {
    36→  if (!date) return null;
    37→  const d = typeof date === 'string' ? new Date(date) : date;
    38→  if (Number.isNaN(d.getTime())) return null;
    39→  const diffMs = Date.now() - d.getTime();
    40→  if (diffMs < 0) return 'Just now';
    41→  const sec = Math.floor(diffMs / 1000);
    42→  if (sec < 60) return 'Just now';
    43→  const min = Math.floor(sec / 60);
    44→  if (min < 60) return `${min}m ago`;
    45→  const hr = Math.floor(min / 60);
    46→  if (hr < 24) return `${hr}h ago`;
    47→  const day = Math.floor(hr / 24);
    48→  if (day < 7) return `${day}d ago`;
    49→  const wk = Math.floor(day / 7);
    50→  if (wk < 4) return `${wk}w ago`;
    51→  // Older than ~4 weeks — fall back to a formatted date
    52→  return d.toLocaleDateString(undefined, {
    53→    month: 'short',
    54→    day: 'numeric',
    55→    year: d.getFullYear() === new Date().getFullYear() ? undefined : 'numeric',
    56→  });
    57→}
